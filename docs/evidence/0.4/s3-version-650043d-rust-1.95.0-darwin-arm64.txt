Appendix: the artifact check (Python 3, run as `python3 -I compare.py OLD NEW`
on the two output directories)

  for name in artifacts:
      old, new = OLD/name bytes, NEW/name bytes
      require new.count(b"0.3.1") == 2 and new.count(b"0.3.0") == 1
      require old.count(b"0.3.0") == 3 and len(old) == len(new)
      require new.replace(b"0.3.1", b"0.3.0") == old
      require the keys whose values differ between OLD/name.json and
          NEW/name.json are {compiler, digest}, or {compiler, digest,
          package} with package.version 0.3.0 -> 0.3.1
      require NEW compiler.version == "0.3.1", language.version == "0.3.0"
